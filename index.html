CROSS-REFERENCE MATRIX (ATOS x SISTEMÁTICA) // ========================================================================= function renderCrossReferenceTable(filteredAtos) { const tbody = document.getElementById(`tbody-cross-reference`); if (!tbody) return; // Exibe todos os atos conforme a aba selecionada (incluindo VIGENTES) let relevant = [...filteredAtos]; if (appState.crossTab === `DESCOBERTO`) { relevant = relevant.filter(a => a.category === `VENCIDO_DESCOBERTO`); } else if (appState.crossTab === `COBERTO`) { // Agora exibe os cursos com respaldo legal garantido: // Cursos plenamente VIGENTES ou com processo ativo no SIPE relevant = relevant.filter(a => a.category === `VIGENTE` || a.matchedProcess !== null); } else if (appState.crossTab === `PREVENTIVO`) { relevant = relevant.filter(a => a.category === `A_VENCER`); } // Ordenação de criticidade: Vencidos primeiro, depois A Vencer, depois Vigentes const rankOrder = { VENCIDO_DESCOBERTO: 1, VENCIDO_COBERTO: 2, A_VENCER: 3, VIGENTE: 4 }; relevant.sort((a, b) => (rankOrder[a.category] - rankOrder[b.category]) || (a.daysRemaining - b.daysRemaining)); if (relevant.length === 0) { tbody.innerHTML = `Nenhum registro encontrado para os filtros selecionados.`; return; } tbody.innerHTML = relevant.map((item, idx) => { const etecObj = ETECS_MASTER.find(e => e.cod === Number(item.cod)); const unitLabel = etecObj ? etecObj.cidade : item.municipio; // Configuração dos Badges de Situação do Ato e Diagnóstico let statusBadge = `A Vencer`; if (item.category === `VIGENTE`) { statusBadge = `Vigente`; } else if (item.daysRemaining !== null && item.daysRemaining < 0) { statusBadge = `Vencido`; } let procHtml = `Sem processo no SIPE`; if (item.matchedProcess) { procHtml = `
Proc. ${escapeHtml(item.matchedProcess.processo)}
${escapeHtml(item.matchedProcess.situacao)} · ${escapeHtml(item.matchedProcess.setor)}
`; } else if (item.category === `VIGENTE`) { procHtml = `Respaldo CEE Pleno`; } let diagBadge = `Monitorar Prazo`; if (item.category === `VENCIDO_DESCOBERTO`) { diagBadge = `Pendente de
Regularização`; } else if (item.category === `VENCIDO_COBERTO`) { diagBadge = `Em Regularização CEE`; } else if (item.category === `VIGENTE`) { diagBadge = `Regularidade Vigente`; } else if (item.category === `A_VENCER` && item.matchedProcess) { diagBadge = `Renovação Protocolada`; } const safeKey = encodeURIComponent(`${item.cod}|${item.municipio}|${item.curso}`); return ` ${escapeHtml(unitLabel)}
Cód. ${item.cod}
${escapeHtml(item.curso)}
${escapeHtml(item.municipio)}
${escapeHtml(item.doc)}
${escapeHtml(item.ato)}
${escapeHtml(item.validade)} ${statusBadge} ${procHtml} ${diagBadge}
Abrir SIPE
`; }).join(``); }
